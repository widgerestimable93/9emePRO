      if (onFailure) onFailure({ success:false, message:'Délai dépassé (' + fnName + ').', code:'TIMEOUT' });
    }, _pendingTimeoutMs);

    var url = GAS_URL
      + '?fn=' + encodeURIComponent(fnName)
      + '&args=' + encodeURIComponent(JSON.stringify(args))
      + '&callback=' + cbName;

    script.src = url;
    script.onerror = function () {
      cleanup();
      if (onFailure) onFailure({ success:false, message:'Erreur réseau.', code:'NETWORK' });
    };
    document.body.appendChild(script);
  }

  function makeRunner(successHandler, failureHandler) {
    return new Proxy({}, {
      get: function (target, fnName) {
        if (fnName === 'withSuccessHandler') return function (cb) { return makeRunner(cb, failureHandler); };
        if (fnName === 'withFailureHandler') return function (cb) { return makeRunner(successHandler, cb); };
        return function () {
          callGAS(fnName, Array.prototype.slice.call(arguments), successHandler, failureHandler);
        };
      }
    });
  }

  window.google = window.google || {};
  window.google.script = window.google.script || {};
  window.google.script.run = makeRunner(null, null);
})();

function loadQuiz() {
  showLoader('Chargement des questions…');
  google.script.run
    .withSuccessHandler(function(res) {
      hideLoader();
      if (!res.success) {
        if (res.code==='PREMIUM_REQUIRED'||res.code==='AUTH') {
          showPage('payment'); showToast(res.message,'error');
        } else showToast(res.message||'Erreur','error');
        return;
      }
      App.questions=res.questions;
      App.quizSessionId=res.quizSessionId;
      App.currentQ=0; App.answers={}; App.scoreLive=0;
      showPage('quiz');
      renderQuestion();
      if (res.tpq>0) startTimer(res.tpq*res.questions.length);
    })
    .withFailureHandler(function(err) { hideLoader(); showToast('Erreur: '+err.message,'error'); })
    .getQuestions({token:App.token, matiere:App.quizMatiere, niveau:App.quizNiveau, limit:App.quizQty});
}

function renderQuestion() {
  var q=App.questions[App.currentQ];
  var body=document.getElementById('quiz-body');
  var saved=App.answers[App.currentQ];
  document.getElementById('quiz-q-current').textContent=App.currentQ+1;
  document.getElementById('quiz-q-total').textContent=App.questions.length;
  document.getElementById('quiz-score-live').textContent=App.scoreLive;
  var pct=((App.currentQ)/App.questions.length*100);
  document.getElementById('quiz-progress-fill').style.width=pct+'%';
  document.getElementById('btn-prev').disabled=App.currentQ===0;
  var isLast=App.currentQ===App.questions.length-1;
  document.getElementById('btn-next').textContent=isLast?'✅ Terminer':'Suivant →';

  var html='<div class="q-matiere-tag">' + (q.matiere) + '</div>\n    <div class="q-text">' + (q.question) + '</div>\n    <div class="choices">';
  var letters=['A','B','C','D'];
  q.choix.forEach(function(c,i) {
    var cls='choice-btn';
    if (saved!==undefined) {
      if (i===saved.correctIndex) cls+=' correct';
      else if (i===saved.userIndex&&i!==saved.correctIndex) cls+=' wrong';
    }
    html+='<button class="'+cls+'" onclick="answerQuestion('+i+')" '+(saved!==undefined?'disabled':'')+' data-idx="'+i+'"><span class="choice-letter">'+letters[i]+'</span>'+c+'</button>';
  });
  html+='</div>';
  // Explication
  if (saved!==undefined&&q.explication) {
    var ex=q.explication;
    html+='<div class="expl-box show">'
        +'<div class="expl-header"><div class="expl-header-left"><span class="expl-icon">💡</span><span class="expl-title">Explication</span></div></div>'
        +'<div class="expl-correct-answer"><span class="expl-ca-label">✅ Bonne réponse</span><span class="expl-ca-text">'+(ex.bonneRep||'')+'</span></div>'
        +'<div class="expl-body">'
        +(ex.resume?'<div class="expl-resume">'+ex.resume+'</div>':'')
        +(ex.points&&ex.points.length?'<div class="expl-points">'+ex.points.map(function(p){return '<div class="expl-point"><span class="expl-point-icon">●</span><span class="expl-point-text">'+p+'</span></div>';}).join('')+'</div>':'')
        +(ex.conseil?'<div class="expl-conseil">💡 '+ex.conseil+'</div>':'')
        +'</div></div>';
  }
  body.innerHTML=html;
}

function answerQuestion(index) {
  if (App.answers[App.currentQ]!==undefined) return;
  var q=App.questions[App.currentQ];
  showLoader('Vérification…');
  google.script.run
    .withSuccessHandler(function(res) {
      hideLoader();
      if (!res.success) { showToast(res.message||'Erreur','error'); return; }
      App.answers[App.currentQ]={ userIndex:index, correctIndex:res.correctIndex, isOk:res.correct, pts:res.pts };
      if (res.correct) { App.scoreLive+=res.pts; showToast('+'+res.pts+' pts 🎉','success'); }
      else showToast('Raté… La bonne réponse s\'affiche.','error');
      renderQuestion();
    })
    .withFailureHandler(function(err) { hideLoader(); showToast('Erreur: '+err.message,'error'); })
    .checkAnswer({quizSessionId:App.quizSessionId, questionId:q.id, answerIndex:index, niveau:q.niveau});
}

function nextQuestion() {
  if (App.currentQ<App.questions.length-1) { App.currentQ++; renderQuestion(); }
  else finishQuiz();
}
function prevQuestion() { if (App.currentQ>0) { App.currentQ--; renderQuestion(); } }

function finishQuiz() {
  if (App.timerInterval) { clearInterval(App.timerInterval); App.timerInterval=null; }
  var answers=Object.keys(App.answers).map(function(k) {
    return {
      questionId: App.questions[parseInt(k)].id,
      answerIndex: App.answers[k].userIndex
    };
  });
  showLoader('Calcul des résultats…');
  google.script.run
    .withSuccessHandler(function(res) {
      hideLoader();
      if (!res.success) { showToast(res.message||'Erreur','error'); return; }
      showPage('result');
      var pct=res.total?Math.round((res.correct/res.total)*100):0;
      var icons=[['🏆',90],['🎓',70],['📈',50],['💪',0]];
      var icon=(icons.find(function(pair) { var t = pair[1]; return pct>=t; })||['💪'])[0];
      document.getElementById('result-icon').textContent=icon;
      document.getElementById('result-title').textContent=pct>=90?'Excellent !!':pct>=70?'Très bien !':pct>=50?'Pas mal !':'Continue !';
      document.getElementById('result-pct').textContent=pct+'%';
      document.getElementById('result-sub').textContent=(res.correct) + '/' + (res.total) + ' correctes — Score total : ' + (res.scoreTotal) + ' pts';
      document.getElementById('res-correct').textContent=res.correct;
      document.getElementById('res-wrong').textContent=res.total-res.correct;
      document.getElementById('res-pts').textContent=res.score;
      App.user.score_total=res.scoreTotal;
      if (res.newBadges&&res.newBadges.length>0) showToast('🏅 Badge obtenu : '+res.newBadges[0].label,'success');
      setTimeout(maybeShowUpgradeModal, 900);
    })
    .withFailureHandler(function(err) { hideLoader(); showToast('Erreur: '+err.message,'error'); })
    .submitQuiz({token:App.token, quizSessionId:App.quizSessionId, answers, matiere:App.quizMatiere, tempsSecondes:0});
}

function startTimer(totalSecs) {
  var remaining=totalSecs;
  var el=document.getElementById('quiz-timer');
  el.style.display='block';
  App.timerInterval=setInterval(function() {
    remaining--;

function loadPaymentPage() {
  // Charger config prix
  google.script.run
    .withSuccessHandler(function(res) {
      if (res.success) {
        App.payConfig=res;
        document.getElementById('pay-price').textContent=res.price+' HTG';
        updateMethodDisplay();
      }
    })
    .withFailureHandler(function() {})
    .getPaymentConfig({});
  // Vérifier statut existant
  if (!App.token) return;
  google.script.run
    .withSuccessHandler(function(res) {
      if (res.success&&res.payment) {
        var p=res.payment;
        var statusSection=document.getElementById('payment-status-section');
        var formSection=document.getElementById('payment-form-section');
        statusSection.style.display='block';
        var badge=document.getElementById('pay-status-badge');
        var detail=document.getElementById('pay-status-detail');
        if (p.status==='pending') {
          badge.className='pstatus-badge pstatus-pending'; badge.textContent='⏳ En attente';
          detail.textContent='Réf. '+p.reference+' — Envoyée le '+p.date.substring(0,10)+'. Validation sous 24h.';
          formSection.style.display='none';
          startPaymentPolling();
        } else if (p.status==='approved') {
          badge.className='pstatus-badge pstatus-approved'; badge.textContent='✅ Approuvé';
          detail.textContent='Premium activé ! Profite de tous tes quiz 🎉';
          formSection.style.display='none';
          App.user.is_premium=true;
          stopPaymentPolling();
        } else if (p.status==='rejected') {
          badge.className='pstatus-badge pstatus-rejected'; badge.textContent='❌ Rejeté';
          var note=p.admin_note?(' Note : '+p.admin_note):'';
          detail.textContent='Ta demande a été rejetée.'+note+' Soumets une nouvelle demande.';
          stopPaymentPolling();
        }
      } else {
        stopPaymentPolling();
      }
    })
    .withFailureHandler(function() {})
    .getMyPaymentStatus({token:App.token});
}

// ── Vérifie en arrière-plan si la demande en attente a été traitée,
// pour activer le Premium "en live" sans que l'étudiant ait à recharger.
App.paymentPollTimer=null;
function startPaymentPolling() {
  if (App.paymentPollTimer) return; // déjà en cours
  App.paymentPollTimer=setInterval(function() {
    google.script.run
      .withSuccessHandler(function(res) {
        if (!res.success||!res.payment) return;
        var p=res.payment;
        if (p.status==='approved') {
          stopPaymentPolling();
          App.user.is_premium=true;
          showToast('🎉 Paiement validé ! Ton accès Premium est activé.','success');
          loadPaymentPage();
        } else if (p.status==='rejected') {
          stopPaymentPolling();
          showToast('❌ Ta demande a été rejetée.','error');
          loadPaymentPage();
        }
      })
      .withFailureHandler(function() {})
      .getMyPaymentStatus({token:App.token});
  }, 10000);
}
function stopPaymentPolling() {
  if (App.paymentPollTimer) { clearInterval(App.paymentPollTimer); App.paymentPollTimer=null; }
}

// Redimensionne l'image côté client (max 900px de large) avant envoi,
// pour éviter d'envoyer des photos de plusieurs Mo au serveur.
App.payPhotoBase64=null; App.payPhotoMime=null;
function onPayPhotoSelected(event) {
  var file=event.target.files[0];
  var statusEl=document.getElementById('pay-photo-status');
  if (!file) return;
  if (!file.type.match(/^image\//)) { statusEl.textContent='Choisis une image.'; return; }
  statusEl.textContent='Traitement de l\'image…';
  var reader=new FileReader();
  reader.onload=function(e) {
    var img=new Image();
    img.onload=function() {
      var maxW=900;
      var scale=Math.min(1, maxW/img.width);
      var canvas=document.createElement('canvas');
      canvas.width=img.width*scale; canvas.height=img.height*scale;
      var ctx=canvas.getContext('2d');
      ctx.drawImage(img,0,0,canvas.width,canvas.height);
      var dataUrl=canvas.toDataURL('image/jpeg',0.75);
      App.payPhotoBase64=dataUrl;
      App.payPhotoMime='image/jpeg';
      document.getElementById('pay-photo-preview').src=dataUrl;
      document.getElementById('pay-photo-preview').style.display='block';
      statusEl.textContent='✅ Photo prête à être envoyée.';
    };
    img.src=e.target.result;
  };
  reader.readAsDataURL(file);
}

var _aiPageLoaded = false;
var _aiExecUrl    = null;
function loadAIPage() {
  var iframe = document.getElementById('ai-iframe');
  if (!iframe) return;
  if (_aiPageLoaded && _aiExecUrl) { _syncThemeToAI(); return; }
  // window.location.href retourne l'URL sandbox (script.googleusercontent.com),
  // pas l'URL /exec → page blanche. On récupère la vraie URL via backend.
  google.script.run
    .withSuccessHandler(function(res) {
      if (!res || !res.url) { _loadAIFallback(iframe); return; }
      _aiExecUrl = res.url;
      iframe.src = _aiExecUrl + '?page=ai';
      _aiPageLoaded = true;
      iframe.addEventListener('load', function() { _syncThemeToAI(); });
    })
    .withFailureHandler(function() { _loadAIFallback(iframe); })
    .getWebAppUrl();
}
function _loadAIFallback(iframe) {
  var href = window.location.href;
  var base = href.indexOf('?') !== -1 ? href.split('?')[0] : href;
  iframe.src = base + '?page=ai';
  _aiPageLoaded = true;
}
function _syncThemeToAI() {
  var iframe = document.getElementById('ai-iframe');
  if (!iframe || !iframe.contentWindow) return;
  var theme = document.documentElement.getAttribute('data-theme') || 'dark';
  try { iframe.contentWindow.postMessage({type:'SET_THEME', theme:theme}, '*'); } catch(e) {}
}
window.addEventListener('message', function(e) {
  if (e.data && e.data.type === 'AI_GO_BACK') showPage('dashboard');
});
